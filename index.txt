function checkTest() {

    let score = 0;

    for (let i = 1; i <= 5; i++) {

        let answer = document.querySelector(
            'input[name="q' + i + '"]:checked'
        );

        if (answer) {
            score += Number(answer.value);
        }
    }

    let result = document.getElementById("result");

    if (score === 5) {
        result.innerHTML =
            "🎉 Ajoyib! Barcha javoblar to‘g‘ri. Natija: 5/5";
    }
    else if (score >= 3) {
        result.innerHTML =
            "👍 Yaxshi! Natija: " + score + "/5. Mavzuni yana bir bor takrorlang.";
    }
    else {
        result.innerHTML =
            "📚 Natija: " + score + "/5. Xavfsizlik qoidalarini yana bir bor o‘rganing.";
    }
}
